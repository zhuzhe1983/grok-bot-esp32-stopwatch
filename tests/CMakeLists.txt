cmake_minimum_required(VERSION 3.20)
project(grok_bot_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(face STATIC ../components/agent_interaction/agent_interaction.c ../main/demo_controller.c)
target_include_directories(face PUBLIC ../components/agent_interaction/include ../components/agent_interaction ../main)
target_compile_options(face PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(face PUBLIC -fsanitize=address,undefined)
if(NOT APPLE)
    target_link_libraries(face PUBLIC m)
endif()
foreach(name interaction controller)
    add_executable(test_${name} test_${name}.c)
    target_link_libraries(test_${name} PRIVATE face)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined)
    add_test(NAME ${name} COMMAND test_${name} "${CMAKE_CURRENT_BINARY_DIR}")
endforeach()
add_executable(render_preview render_preview.c)
target_link_libraries(render_preview PRIVATE face)
