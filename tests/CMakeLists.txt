cmake_minimum_required(VERSION 3.20)
project(grok_bot_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(face STATIC ../components/agent_interaction/agent_interaction.c ../main/demo_controller.c ../main/raster_buffer.c)
target_include_directories(face PUBLIC ../components/agent_interaction/include ../components/agent_interaction ../main)
target_compile_options(face PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(face PUBLIC -fsanitize=address,undefined)
if(NOT APPLE)
    target_link_libraries(face PUBLIC m)
endif()
foreach(name interaction controller)
    add_executable(test_${name} test_${name}.c)
    target_link_libraries(test_${name} PRIVATE face)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined)
    add_test(NAME ${name} COMMAND test_${name} "${CMAKE_CURRENT_BINARY_DIR}")
endforeach()
add_library(reference_renderer STATIC reference/renderer_0_1_1.c)
target_include_directories(reference_renderer PRIVATE ../components/agent_interaction/include ../components/agent_interaction)
foreach(symbol init configure set_state input level look animated step pcm_level render)
    target_compile_definitions(reference_renderer PRIVATE ea_interaction_${symbol}=reference_${symbol})
endforeach()
target_compile_definitions(reference_renderer PRIVATE eai_project_eye=reference_project_eye eai_project_surface_eye=reference_project_surface_eye)
target_compile_options(reference_renderer PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined)
add_executable(test_raster test_raster.c)
target_link_libraries(test_raster PRIVATE face reference_renderer)
target_compile_options(test_raster PRIVATE -Wall -Wextra -Werror -UNDEBUG -fsanitize=address,undefined)
add_test(NAME raster COMMAND test_raster)
add_executable(render_preview render_preview.c)
target_link_libraries(render_preview PRIVATE face)
